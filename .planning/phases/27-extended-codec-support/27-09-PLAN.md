/// Fluent builder for creating video DICOM files.
///
/// Example usage:
///   var file = new VideoDicomBuilder()
///       .WithSopClass(VideoSopClass.Endoscopic)
///       .WithTransferSyntax(TransferSyntax.H264HighProfile41)
///       .WithPatientFromTemplate(existingDataset)
///       .WithFrameRate(30.0)
///       .WithPixelData(encodedVideoBytes)
///       .Build();
public sealed class VideoDicomBuilder
{
    private VideoSopClass? _sopClass;
    private TransferSyntax _transferSyntax = TransferSyntax.H264HighProfile41;
    private DicomDataset? _templateDataset;
    private double _frameRate;
    private int _width;
    private int _height;
    private byte[]? _encodedVideoData;
    private string? _sopInstanceUid;
    private string? _seriesInstanceUid;
    private string? _studyInstanceUid;
    private string? _patientId;
    private string? _patientName;
    // ... other metadata fields

    public VideoDicomBuilder WithSopClass(VideoSopClass sopClass);
    public VideoDicomBuilder WithTransferSyntax(TransferSyntax ts);
    public VideoDicomBuilder WithPatientFromTemplate(DicomDataset template);
    public VideoDicomBuilder WithFrameRate(double fps);
    public VideoDicomBuilder WithDimensions(int width, int height);
    public VideoDicomBuilder WithPixelData(byte[] encodedVideoData);
    public VideoDicomBuilder WithSopInstanceUid(string uid);
    public VideoDicomBuilder WithSeriesInstanceUid(string uid);
    public VideoDicomBuilder WithStudyInstanceUid(string uid);
    public VideoDicomBuilder WithPatient(string patientId, string patientName);

    public DicomFile Build();
}
```

Build() method:
1. Validate required fields: sopClass, encodedVideoData, width, height, frameRate > 0
2. Map VideoSopClass to SOP Class UID string (using the video SOP class UIDs from Plan 01)
3. Auto-generate UIDs using "2.25." + UUID format if not provided:
   - SOPInstanceUID
   - SeriesInstanceUID
   - StudyInstanceUID
4. Create DicomDataset with required DICOM attributes:
   - (0008,0016) SOPClassUID
   - (0008,0018) SOPInstanceUID
   - (0020,000D) StudyInstanceUID
   - (0020,000E) SeriesInstanceUID
   - (0008,0060) Modality = derive from SOP class (ES, SM, XC, XA, XRF, US, SC)
   - (0028,0010) Rows = height
   - (0028,0011) Columns = width
   - (0028,0100) BitsAllocated = 8
   - (0028,0101) BitsStored = 8
   - (0028,0102) HighBit = 7
   - (0028,0103) PixelRepresentation = 0 (unsigned)
   - (0028,0004) PhotometricInterpretation = "YBR_FULL_422" or "YBR_PARTIAL_420" depending on codec
   - (0028,0008) NumberOfFrames = -1 or computed if known
   - (0028,0002) SamplesPerPixel = 3 (or 1 for monochrome)
   - (0018,1063) FrameTime = 1000.0 / frameRate
   - (0018,0040) CineRate = (int)frameRate
5. If templateDataset provided, copy patient/study-level attributes:
   - Patient: (0010,0010), (0010,0020), (0010,0030), (0010,0040)
   - Study: (0008,0020), (0008,0030), (0008,0050)
6. Create encapsulated pixel data as single fragment containing the video bitstream:
   - Create DicomFragmentSequence with the encodedVideoData as a single fragment
   - Per context decision: single encapsulated fragment, not per-GOP
7. Warn if encoded data size > 2^32 - 2 bytes (4GB limit per RESEARCH.md Pitfall 6)
8. Create DicomFile with the dataset and selected transfer syntax

/// Encodes frames to video and creates a complete DICOM file.
public static async Task<DicomFile> CreateVideoDicomAsync(
    IAsyncEnumerable<VideoFrame> frames,
    VideoEncoderOptions options,
    int width, int height,
    VideoSopClass sopClass,
    DicomDataset? templateDataset = null,
    IProgress<VideoEncodeProgress>? progress = null,
    CancellationToken ct = default)
{
    byte[] encodedVideo = await EncodeFromFramesAsync(frames, options, width, height, progress, ct);

    var builder = new VideoDicomBuilder()
        .WithSopClass(sopClass)
        .WithTransferSyntax(MapCodecToTransferSyntax(options.Codec))
        .WithDimensions(width, height)
        .WithFrameRate(options.FrameRate)
        .WithPixelData(encodedVideo);

    if (templateDataset != null)
        builder.WithPatientFromTemplate(templateDataset);

    return builder.Build();
}
```
