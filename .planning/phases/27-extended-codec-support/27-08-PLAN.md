```csharp
/// Container for a single video frame's pixel data.
public sealed class VideoFrame : IDisposable
{
    public ReadOnlyMemory<byte> PixelData { get; }
    public int Width { get; }
    public int Height { get; }
    public VideoPixelFormat Format { get; }
    // Optional audio samples for this frame's time window
    public ReadOnlyMemory<byte>? AudioSamples { get; init; }
    public AudioSampleFormat AudioFormat { get; init; }

    public VideoFrame(ReadOnlyMemory<byte> pixelData, int width, int height, VideoPixelFormat format);
    public void Dispose();
}

public enum VideoPixelFormat { Rgb24, Gray8, Gray16, Yuv420P }
public enum AudioSampleFormat { None, Pcm16, Float32 }
```

2. Create VideoEncoderOptions.cs per RESEARCH.md Pattern 3:
```csharp
public record VideoEncoderOptions
{
    public VideoCodecType Codec { get; init; } = VideoCodecType.H264;
    public VideoQualityPreset Preset { get; init; } = VideoQualityPreset.Diagnostic;
    public double FrameRate { get; init; } = 30.0;
    public AudioCodecType AudioCodec { get; init; } = AudioCodecType.None;
    public int AudioSampleRate { get; init; } = 48000;
    public int AudioChannels { get; init; } = 2;
    public HardwareAcceleration HwAccel { get; init; } = HardwareAcceleration.Auto;
    public int GopSize { get; init; }  // 0 = codec default
    /// Escape hatch for raw encoder parameters (CRF, bitrate, etc.)
    public Dictionary<string, string>? RawParameters { get; init; }

    // Preset factories
    public static VideoEncoderOptions Diagnostic => new() { Preset = VideoQualityPreset.Diagnostic };
    public static VideoEncoderOptions Review => new() { Preset = VideoQualityPreset.Review };
    public static VideoEncoderOptions Archive => new() { Preset = VideoQualityPreset.Archive };
}

public enum VideoCodecType { MPEG2, H264, HEVC }
public enum VideoQualityPreset { Diagnostic, Review, Archive }
public enum AudioCodecType { None, AAC, PCM }
public enum HardwareAcceleration { Auto, ForceCpu, PreferGpu }
```
