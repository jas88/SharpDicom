```c
typedef struct video_encoder video_encoder_t;

typedef struct {
    int codec_id;           // VIDEO_CODEC_*
    int width;
    int height;
    double frame_rate;
    int bit_depth;          // 8 or 10
    int gop_size;           // keyframe interval (0 = codec default)
    int quality_preset;     // VIDEO_QUALITY_*
    int crf;                // -1 for preset default
    int bitrate;            // target bps (0 = CRF mode)
```

Create video_encoder.c implementing the API:

When SHARPDICOM_WITH_FFMPEG_ENC is defined (vendor libs present):

video_encoder_create:
1. Allocate video_encoder_t struct
2. Select encoder based on codec_id and hw_accel:
   - For H264: try h264_nvenc, h264_videotoolbox, h264_vaapi (if hw_accel != CPU), fallback to libx264
   - For HEVC: try hevc_nvenc, hevc_videotoolbox, hevc_vaapi, fallback to libx265
   - For MPEG2: always use mpeg2video (no GPU encoder)
3. Create AVCodecContext with parameters from config
4. Apply quality preset:
   - Diagnostic: H264 CRF=17, HEVC CRF=20, MPEG2 bitrate=15Mbps
   - Review: H264 CRF=23, HEVC CRF=26, MPEG2 bitrate=8Mbps
   - Archive: H264 CRF=28, HEVC CRF=31, MPEG2 bitrate=4Mbps
5. If crf >= 0, override preset CRF
6. If bitrate > 0, use bitrate mode instead of CRF
7. Open codec via avcodec_open2
8. Create SwsContext for pixel format conversion (input format -> YUV420P)
9. If audio_codec != NONE, create audio encoder context (AAC or PCM)
10. Create AVFormatContext for muxing (using Annex-B format for H264/HEVC raw bitstream output)

video_encode_frame:
1. Convert input pixels via sws_scale to YUV420P
2. Set pts based on frame count
3. Call avcodec_send_frame
4. Call avcodec_receive_packet in loop
5. Write packets to muxer context
6. Set *packet_available = 1 if any packets produced

video_encode_audio:
1. Create AVFrame with audio samples
2. Convert format via swr_convert if needed
3. Send to audio encoder
4. Interleave with video via av_interleaved_write_frame

video_encoder_flush:
1. Send NULL frame to avcodec_send_frame (flush signal)
2. Drain remaining packets
3. Write trailer

video_encoder_get_output:
1. Return the accumulated output buffer
2. For in-memory muxing, use avio_open_dyn_buf / avio_close_dyn_buf

video_encoder_destroy:
1. Free SwsContext, AVCodecContext, AVFormatContext
2. Free all allocated buffers

When SHARPDICOM_WITH_FFMPEG_ENC is NOT defined:
All functions return SHARPDICOM_ERR_UNSUPPORTED (stub mode).

Error handling: use sharpdicom_set_error() for all error paths with descriptive messages.

```c
#ifndef STB_IMAGE_WRAPPER_H
#define STB_IMAGE_WRAPPER_H

#include "sharpdicom_codecs.h"

#ifdef __cplusplus
extern "C" {
#endif

/**
 * Load an image from memory buffer.
 * Supports: JPEG, PNG, BMP, TGA, PSD, GIF, HDR, PIC, PNM
 * Note: TIFF is NOT supported by stb_image.
 *
 * @param buffer          Input image file data
 * @param len             Length of input data
 * @param pixels_out      Receives pointer to decoded pixel data (call stbi_free_wrapper to free)
 * @param width           Receives image width
 * @param height          Receives image height
 * @param channels        Receives actual channel count
 * @param desired_channels Desired output channels (0=same as source, 1=gray, 3=RGB, 4=RGBA)
 * @return 0 on success, negative error code on failure
 */
SHARPDICOM_API int stbi_load_from_memory_wrapper(
    const uint8_t* buffer, int len,
    uint8_t** pixels_out, int* width, int* height, int* channels,
    int desired_channels);

/**
 * Free pixel data allocated by stbi_load_from_memory_wrapper.
 * @param pixels Pointer returned by stbi_load_from_memory_wrapper (may be NULL)
 */
SHARPDICOM_API void stbi_free_wrapper(uint8_t* pixels);

#ifdef __cplusplus
}
#endif

#endif /* STB_IMAGE_WRAPPER_H */
```

3. Create native/src/stb_image_wrapper.c:
```c
#ifdef SHARPDICOM_WITH_STB_IMAGE
#define STB_IMAGE_IMPLEMENTATION
#define STBI_NO_STDIO          /* No file I/O, memory only */
#define STBI_NO_HDR            /* Skip HDR format (not needed for medical imaging) */
#include "../vendor/stb/stb_image.h"
#endif

#include "stb_image_wrapper.h"
#include "sharpdicom_codecs.h"

int stbi_load_from_memory_wrapper(
    const uint8_t* buffer, int len,
    uint8_t** pixels_out, int* width, int* height, int* channels,
    int desired_channels)
{
#ifdef SHARPDICOM_WITH_STB_IMAGE
    if (!buffer || len <= 0 || !pixels_out || !width || !height || !channels) {
        return SHARPDICOM_ERR_INVALID_ARGUMENT;
    }
    *pixels_out = stbi_load_from_memory(buffer, len, width, height, channels, desired_channels);
    if (!*pixels_out) {
        /* stbi_failure_reason() returns error string */
        return SHARPDICOM_ERR_DECODE_FAILED;
    }
    return SHARPDICOM_OK;
#else
    (void)buffer; (void)len; (void)pixels_out; (void)width; (void)height;
    (void)channels; (void)desired_channels;
    return SHARPDICOM_ERR_UNSUPPORTED;
#endif
}

void stbi_free_wrapper(uint8_t* pixels)
{
#ifdef SHARPDICOM_WITH_STB_IMAGE
    if (pixels) {
        stbi_image_free(pixels);
    }
#else
    (void)pixels;
#endif
}
```
