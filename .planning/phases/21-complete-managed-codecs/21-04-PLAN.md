```csharp
public enum CodecErrorMode
{
    Strict,   // Fail fast on any error
    Lenient   // Best-effort decode, return partial data
}

// In JpegLsDecoder
public static DecodeResult TryDecode(
    ReadOnlySpan<byte> data,
    PixelDataInfo info,
    Span<byte> destination,
    int frameIndex,
    CodecErrorMode errorMode = CodecErrorMode.Strict)
{
    // On truncated stream:
    if (errorMode == CodecErrorMode.Lenient)
    {
        // Return partial decode with warning in result
        return DecodeResult.Partial(bytesDecoded, "Stream truncated at offset ...");
    }
    else
    {
        return DecodeResult.Fail(frameIndex, position, "Unexpected end of stream");
    }
}
```
