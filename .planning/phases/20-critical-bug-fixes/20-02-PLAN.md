```csharp
private static async Task<DicomDataset> ParseDatasetAsync(
    byte[] data,
    DicomAssociation association,
    byte contextId,
    CancellationToken ct)
{
    // Get the accepted transfer syntax for this presentation context
    var context = association.GetPresentationContext(contextId);
    var transferSyntax = context?.AcceptedTransferSyntax ?? TransferSyntax.ImplicitVRLittleEndian;

    // Use DicomFileReader for complete parsing including sequences
    using var ms = new MemoryStream(data);
    var options = new DicomReaderOptions
    {
        TransferSyntax = transferSyntax,
        // Skip pixel data if in streaming mode - handler receives it separately
        PixelDataHandling = PixelDataHandling.LoadInMemory
    };

    using var reader = new DicomFileReader(ms, options, leaveOpen: true);

    // Skip file meta info - P-DATA contains raw dataset, not Part 10 file
    // The data is already in the negotiated transfer syntax
    reader.SetTransferSyntax(transferSyntax);

    // Read dataset directly (not Part 10 format)
    return await reader.ReadDatasetAsync(ct).ConfigureAwait(false);
}
```
