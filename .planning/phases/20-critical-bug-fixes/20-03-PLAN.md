using FsCheck;
using FsCheck.NUnit;
using NUnit.Framework;

namespace SharpDicom.Tests.IO;

[TestFixture]
public class PropertyBasedSequenceTests
{
    // Property 1: Roundtrip preserves all elements
    [FsCheck.NUnit.Property]
    public Property Roundtrip_DefinedLength_PreservesAllElements()
    {
        return Prop.ForAll(
            NestedSequenceArbitrary.Generate(),
            data =>
            {
                var dataset = data.ToDataset();
                var options = new DicomWriterOptions { SequenceLength = SequenceLengthEncoding.Defined };

                using var ms = new MemoryStream();
                var file = new DicomFile(dataset);
                file.SaveAsync(ms, options).GetAwaiter().GetResult();

                ms.Position = 0;
                var roundtrip = DicomFile.OpenAsync(ms).GetAwaiter().GetResult();

                return DeepEquals(dataset, roundtrip.Dataset)
                    .Label("Element equality after roundtrip");
            });
    }

    // Property 2: Roundtrip with undefined length also preserves elements
    [FsCheck.NUnit.Property]
    public Property Roundtrip_UndefinedLength_PreservesAllElements()
    {
        return Prop.ForAll(
            NestedSequenceArbitrary.Generate(),
            data =>
            {
                var dataset = data.ToDataset();
                var options = new DicomWriterOptions { SequenceLength = SequenceLengthEncoding.Undefined };

                using var ms = new MemoryStream();
                var file = new DicomFile(dataset);
                file.SaveAsync(ms, options).GetAwaiter().GetResult();

                ms.Position = 0;
                var roundtrip = DicomFile.OpenAsync(ms).GetAwaiter().GetResult();

                return DeepEquals(dataset, roundtrip.Dataset)
                    .Label("Element equality after undefined-length roundtrip");
            });
    }

    // Property 3: Depth tracking never goes negative
    [FsCheck.NUnit.Property]
    public Property FindSequenceDelimiter_DepthNeverNegative()
    {
        return Prop.ForAll(
            NestedSequenceArbitrary.GenerateBytes(maxDepth: 10),
            bytes =>
            {
                var reader = new DicomStreamReader(bytes.AsSpan(), true, true);
                // Property: Method completes without throwing depth underflow
                var result = reader.FindSequenceDelimiter();
                return (result >= 0 || result == -1).Label("Valid result or not found");
            });
    }

    // Generator for nested sequence test data
    private static class NestedSequenceArbitrary
    {
        public static Arbitrary<NestedSequenceData> Generate()
        {
            return Arb.From(GenNestedSequence(3, maxDepth: 5));
        }

        public static Arbitrary<byte[]> GenerateBytes(int maxDepth)
        {
            return Arb.From(GenNestedSequenceBytes(maxDepth));
        }

        private static Gen<NestedSequenceData> GenNestedSequence(int size, int maxDepth, int depth = 0)
        {
            if (depth >= maxDepth || size <= 0)
                return Gen.Constant(new NestedSequenceData());

            return from itemCount in Gen.Choose(0, Math.Min(2, size))
                   from items in Gen.ListOf(itemCount, GenItem(size / 2, maxDepth, depth + 1))
                   select new NestedSequenceData { Items = items.ToList() };
        }

        private static Gen<ItemData> GenItem(int size, int maxDepth, int depth)
        {
            return from elementCount in Gen.Choose(1, 3)
                   from nested in GenNestedSequence(size / 2, maxDepth, depth)
                   select new ItemData
                   {
                       ElementCount = elementCount,
                       NestedSequence = nested
                   };
        }
    }

    // Data structures for property tests

    }

public class DcmtkInteropTests
{
    private static bool IsDcmtkAvailable()
    {
        try
        {
            var psi = new ProcessStartInfo("dcmdump", "--version")
            {
                RedirectStandardOutput = true,
                UseShellExecute = false
            };
            using var p = Process.Start(psi);
            p?.WaitForExit(5000);
            return p?.ExitCode == 0;
        }
        catch
        {
            return false;
        }
    }

    private static ProcessResult RunDcmtk(string tool, string args)
    {
        var psi = new ProcessStartInfo(tool, args)
        {
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false
        };

        using var p = Process.Start(psi);
        p.WaitForExit(30000);

        return new ProcessResult
        {
            ExitCode = p.ExitCode,
            Stdout = p.StandardOutput.ReadToEnd(),
            Stderr = p.StandardError.ReadToEnd()
