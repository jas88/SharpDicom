/// <summary>
/// De-identification action codes per DICOM PS3.15 Section E.1.
/// </summary>
public enum DeidentificationAction : byte
{
    /// <summary>None - no action defined, attribute not in standard profile.</summary>
    None = 0,
    /// <summary>D - replace with dummy value consistent with VR (Type 1/2 safe).</summary>
    Dummy = (byte)'D',
    /// <summary>Z - replace with zero-length value (Type 2/3 only).</summary>
    Zero = (byte)'Z',
    /// <summary>X - remove entirely.</summary>
    Remove = (byte)'X',
    /// <summary>K - keep unchanged (clean nested sequences).</summary>
    Keep = (byte)'K',
    /// <summary>C - clean - replace with non-identifying value preserving meaning.</summary>
    Clean = (byte)'C',
    /// <summary>U - replace UID with consistent remapped UID.</summary>
    UidRemap = (byte)'U'
}

Create stateful context for multi-file de-identification:
```csharp
namespace SharpDicom.Deidentification;

/// <summary>
/// Tracks UID mappings and date offsets for consistent de-identification across a study/batch.
/// Thread-safe for concurrent multi-file processing.
/// </summary>
public sealed class DeidentificationContext : IDisposable
{
    private readonly ConcurrentDictionary<DicomUID, DicomUID> _uidMap = new();
    private readonly ConcurrentDictionary<string, TimeSpan> _patientDateOffsets = new();
    private readonly ConcurrentDictionary<DicomUID, TimeSpan> _studyDateOffsets = new();
    private readonly string _uidPrefix;
    private readonly (int Min, int Max) _dateShiftRange;
    private readonly DateShiftStrategy _strategy;

    public DeidentificationContext(DeidentificationOptions options)
    {
        _uidPrefix = options.UidPrefix;
        _dateShiftRange = options.DateShiftRange;
        _strategy = options.DateShiftStrategy;
    }

    /// <summary>Gets or creates a remapped UID for the original UID.</summary>
    public DicomUID RemapUID(DicomUID original)
    {
        return _uidMap.GetOrAdd(original, _ => DicomUID.Generate());
    }

    /// <summary>Gets the date offset for a patient (creates if not exists).</summary>
    public TimeSpan GetDateOffset(string patientId)
    {
        return _patientDateOffsets.GetOrAdd(patientId, _ => CreateRandomOffset());
    }

    /// <summary>Gets the date offset for a study (creates if not exists).</summary>
    public TimeSpan GetStudyDateOffset(DicomUID studyInstanceUid)
    {
        return _studyDateOffsets.GetOrAdd(studyInstanceUid, _ => CreateRandomOffset());
    }

    private TimeSpan CreateRandomOffset()
    {
        var days = Random.Shared.Next(_dateShiftRange.Min, _dateShiftRange.Max + 1);
        return TimeSpan.FromDays(days);
    }

    /// <summary>Gets all UID mappings (for serialization/audit).</summary>
    public IReadOnlyDictionary<string, string> GetUidMappings()
    {
        return _uidMap.ToDictionary(
            kvp => kvp.Key.ToString(),
            kvp => kvp.Value.ToString());
    }

    /// <summary>Saves context to stream for persistence between sessions.</summary>
    public async Task SaveAsync(Stream stream, CancellationToken ct = default)
    {
        var data = new ContextData
        {
            UidMappings = _uidMap.ToDictionary(k => k.Key.ToString(), v => v.Value.ToString()),
            PatientOffsets = _patientDateOffsets.ToDictionary(k => k.Key, v => v.Value.TotalDays),
            StudyOffsets = _studyDateOffsets.ToDictionary(k => k.Key.ToString(), v => v.Value.TotalDays)
        };
        await JsonSerializer.SerializeAsync(stream, data, cancellationToken: ct);
    }

    /// <summary>Loads context from stream.</summary>
    public static async Task<DeidentificationContext> LoadAsync(
        Stream stream, DeidentificationOptions options, CancellationToken ct = default)
    {
        var context = new DeidentificationContext(options);
        var data = await JsonSerializer.DeserializeAsync<ContextData>(stream, cancellationToken: ct);
        if (data != null)
        {
            foreach (var (k, v) in data.UidMappings)
                context._uidMap[new DicomUID(k)] = new DicomUID(v);
            foreach (var (k, v) in data.PatientOffsets)
                context._patientDateOffsets[k] = TimeSpan.FromDays(v);
            foreach (var (k, v) in data.StudyOffsets)
                context._studyDateOffsets[new DicomUID(k)] = TimeSpan.FromDays(v);
        }
        return context;
    }

    public void Dispose() { /* No unmanaged resources */ }

    private sealed class ContextData
    {
        public Dictionary<string, string> UidMappings { get; set; } = new();
        public Dictionary<string, double> PatientOffsets { get; set; } = new();
        public Dictionary<string, double> StudyOffsets { get; set; } = new();
    }
}
```

Use System.Text.Json for serialization (built-in, no dependencies).
Use ConcurrentDictionary for thread safety during parallel batch processing.
