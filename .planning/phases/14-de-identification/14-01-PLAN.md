```csharp
// DeidentificationActionTable.Generated.cs
namespace SharpDicom.Deidentification;

public static partial class DeidentificationActionTable
{
    private static readonly FrozenDictionary<uint, ActionEntry> _actions =
        new Dictionary<uint, ActionEntry>
        {
            // (0008,0050) AccessionNumber: Basic=X, RetainUIDs=X, ...
            [0x00080050] = new ActionEntry(X, X, X, X, X, X, X, X, X, X),
            // ... more entries
        }.ToFrozenDictionary();

    public static DeidentificationAction GetAction(DicomTag tag, DeidentificationProfile profile)
    {
        if (!_actions.TryGetValue(tag.ToUInt32(), out var entry))
            return DeidentificationAction.X; // Default: remove unknown
        return entry.GetAction(profile);
    }
}

internal readonly record struct ActionEntry(
    DeidentificationAction Basic,
    DeidentificationAction RetainSafePrivate,
    DeidentificationAction RetainUIDs,
    DeidentificationAction RetainDeviceIdentity,
    DeidentificationAction RetainPatientCharacteristics,
    DeidentificationAction RetainLongModifDates,
    DeidentificationAction RetainLongFullDates,
    DeidentificationAction CleanDesc,
    DeidentificationAction CleanStructuredContent,
    DeidentificationAction CleanGraph)
{
    public DeidentificationAction GetAction(DeidentificationProfile profile) => profile switch
    {
        DeidentificationProfile.Basic => Basic,
        // ... other profiles
        _ => Basic
    };
}
```
